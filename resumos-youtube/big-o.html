Resumo — "A Base das Entrevistas de Algoritmo (Big O)"

Fonte: youtube.com/watch?v=JegDr-LFaTY · pt-BR · ~6:52 · 30/08/2026

Tese central: o mesmo código que parece instantâneo com 100 usuários pode travar com 1 milhão. Big O enxerga qual operação cresce junto com os dados antes de escolher estrutura e algoritmo.

Pontos-chave

1. Estrutura ≠ algoritmo. Estrutura = como os dados ficam guardados; algoritmo = sequência de passos. Andam juntos: a melhor estrutura torna o algoritmo bem mais direto.

2. Pergunte: qual operação priorizar? Buscar, inserir, remover, percorrer, manter ordem, consultar por chave. Cada estrutura favorece umas e deixa outras caras.

3. Índice por e-mail. Lista = testa um por um (O(n)); índice = vai direto na chave (O(1)). Às vezes a melhoria vem de guardar os dados de outro jeito, não dos passos.

4. Milissegundos enganam. Mudam com máquina, linguagem, banco, cache. A pergunta certa: o que acontece quando o dado cresce?

5. Crescimento. Loop = O(n); dois loops aninhados = O(n²). Com 100.000 itens, O(n²) fica inviável.

6. Space–time tradeoff. Gastar memória (índice por fora) economiza passos.

7. 4 perguntas. Quem é o N? Qual a operação mais comum? Qual estrutura ajuda? Como o custo cresce?

8. Melhor/pior/médio caso. Entrevista = pior caso; runtime = caso médio.

9. Sinais. "Duplicados" → sem comparar todos × todos; "chave" → O(1); "próximo/menor/caminho/prefixo" → famílias específicas.

10. Big O não é dogma. Menor notação ≠ sempre melhor; depende do limite de memória/tempo e do volume.

Exemplos de código — busca com e sem índice

3 exemplos, validação real (Node v24.19.0, OpenJDK 26.0.2.1, Python 3.14.7 — saídas equivalentes).

1. Busca linear O(n) vs busca binária O(log n)

// JavaScript (Node) — O(n) vs O(log n)
const contas = Array.from(
  { length: 100000 },
  (_, i) => ({ id: i,
    email: `user${i}@x.com` }));

// O(n): percorre até achar
function buscaLinear(lista, email) {
  for (const c of lista)
    if (c.email === email) return c;
  return null;
}

// O(log n): descarta metade a cada passo
// (lista ORDENADA por email)
function buscaBinaria(lista, email) {
  let lo = 0, hi = lista.length - 1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    const e = lista[mid].email;
    if (e === email) return lista[mid];
    if (e < email) lo = mid + 1;
    else hi = mid - 1;
  }
  return null;
}

const alvo = 'user99999@x.com';
const t0 = performance.now();
const r1 = buscaLinear(contas, alvo);
const t1 = performance.now();
contas.sort((a, b) =>
  a.email.localeCompare(b.email));
const r2 = buscaBinaria(contas, alvo);
const t2 = performance.now();
console.log({ linear: Math.round(t1 - t0),
  binaria: Math.round(t2 - t1),
  achou: r2 !== null });
// Java 26 — O(n) vs O(log n)
import java.util.*;

public class BuscaDemo {
  record Conta(int id,
    String email) {}

  static Conta linear(
    List<Conta> l, String em) {
    for (Conta c : l)
      if (c.email.equals(em))
        return c;
    return null;
  }

  static Conta binaria(
    List<Conta> l, String em) {
    int lo = 0, hi = l.size() - 1;
    while (lo <= hi) {
      int mid = (lo + hi) >>> 1;
      int cmp =
        l.get(mid).email.compareTo(em);
      if (cmp == 0)
        return l.get(mid);
      if (cmp < 0) lo = mid + 1;
      else hi = mid - 1;
    }
    return null;
  }

  public static void main(
    String[] a) {
    var l = new ArrayList<Conta>();
    for (int i = 0; i < 100_000; i++)
      l.add(new Conta(i,
        "user" + i + "@x.com"));
    String alvo =
      "user99999@x.com";
    long t0 = System.nanoTime();
    var r1 = linear(l, alvo);
    long t1 = System.nanoTime();
    l.sort(Comparator
      .comparing(Conta::email));
    var r2 = binaria(l, alvo);
    long t2 = System.nanoTime();
    System.out.println(
      "linear  : " +
      (t1 - t0) / 1_000_000 +
      " ms");
    System.out.println(
      "binaria : " +
      (t2 - t1) / 1_000_000 +
      " ms");
    System.out.println(
      "achou   : " +
      (r2 != null));
  }
}
# Python 3.14 — O(n) vs O(log n)
import time

contas = [{"id": i,
  "email": f"user{i}@x.com"}
  for i in range(100_000)]

def linear(lista, email):
    for c in lista:
        if c["email"] == email:
            return c
    return None

def binaria(lista, email):
    lo, hi = 0, len(lista) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        e = lista[mid]["email"]
        if e == email:
            return lista[mid]
        if e < email: lo = mid + 1
        else: hi = mid - 1
    return None

alvo = "user99999@x.com"
t0 = time.perf_counter()
r1 = linear(contas, alvo)
t1 = time.perf_counter()
contas.sort(key=lambda c:
  c["email"])
r2 = binaria(contas, alvo)
t2 = time.perf_counter()
print(f"linear  : {round((t1-t0)*1000)} ms")
print(f"binaria : {round((t2-t1)*1000)} ms")
print("achou   :",
  r2 is not None)

2. Dois loops aninhados O(n²)

// JavaScript — O(n²)
// achar duplicados SEM estrutura auxiliar
function duplicados(l) {
  const out = [];
  for (let i = 0; i < l.length; i++)
    for (let j = i + 1;
      j < l.length; j++)
      if (l[i] === l[j]
          && !out.includes(l[i]))
        out.push(l[i]);
  return out;
}
console.log(duplicados(
  [3, 1, 4, 1, 5, 9, 2, 6, 5]));
// [ 1, 5 ]
// Java 26 — O(n²)
import java.util.*;

public class O2Demo {
  static List<Integer>
    dup(List<Integer> l) {
    var out =
      new ArrayList<Integer>();
    for (int i = 0;
      i < l.size(); i++)
      for (int j = i + 1;
        j < l.size(); j++)
        if (l.get(i).equals(l.get(j))
            && !out.contains(l.get(i)))
          out.add(l.get(i));
    return out;
  }
  public static void main(
    String[] a) {
    System.out.println(dup(
      List.of(3, 1, 4, 1, 5, 9,
        2, 6, 5)));
  }
}
# Python 3.14 — O(n²)
def duplicados(l):
    out = []
    for i in range(len(l)):
        for j in range(i + 1,
          len(l)):
            if (l[i] == l[j]
                and l[i]
                not in out):
                out.append(l[i])
    return out

print(duplicados(
  [3, 1, 4, 1, 5, 9, 2, 6, 5]))
# [1, 5]

3. Índice por fora — O(n) → O(1)

// JavaScript — hash index
const contas = [
  { id: 1, email: 'ana@x.com' },
  { id: 2, email: 'bia@x.com' },
  { id: 3, email: 'caio@x.com' },
];
const idx = new Map();
for (const c of contas)
  idx.set(c.email, c);

console.log(idx.get(
  'bia@x.com'));
// { id: 2, email: 'bia@x.com' }
// sem índice: for O(n)
// Java 26 — HashMap como índice
import java.util.*;

public class IndexDemo {
  record Conta(int id,
    String email) {}
  public static void main(
    String[] a) {
    var contas = List.of(
      new Conta(1,
        "ana@x.com"),
      new Conta(2,
        "bia@x.com"),
      new Conta(3,
        "caio@x.com"));
    var idx =
      new HashMap<String,
        Conta>();
    for (var c : contas)
      idx.put(c.email(), c);
    System.out.println(
      idx.get("bia@x.com"));
  }
}
# Python 3.14 — dict como índice
contas = [
  {"id": 1,
    "email": "ana@x.com"},
  {"id": 2,
    "email": "bia@x.com"},
  {"id": 3,
    "email": "caio@x.com"},
]
# índice por e-mail
idx = {c["email"]: c
  for c in contas}
print(idx["bia@x.com"])
# {'id': 2, 'email': 'bia@x.com'}
