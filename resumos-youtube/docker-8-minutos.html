Resumo — "Docker Explicado em 8 Minutos"

Fonte: youtube.com/watch?v=jftIzkXbKKY · pt-BR · ~8:14 · 30/08/2026

Tese central: "Na minha máquina funciona" — contêineres eliminam esse problema empacotando a aplicação com tudo que ela precisa, em um ambiente isolado.

Pontos-chave

1. O problema. 3 apps no mesmo servidor compartilhavam SO, bibliotecas, diretórios. Python 2.7 vs 3.11, mesma porta 8080, dev ≠ produção.

2. VM ≠ contêiner. VM = SO inteiro (karne, drivers, pacotes) → pesado, lento. Casa inteira vs quarto de hotel.

3. Namespaces. Ilusão de isolamento: cada contêiner acha que é o único (PID 1, própria rede, próprio FS).

4. Cgroups. Limites de CPU/memória/disco/rede. Um contêiner pesado não derruba os outros (disjuntor por apartamento).

5. Dockerfile. Receita em camadas cacheadas: FROM, WORKDIR, COPY, RUN, CMD. Só o código muda → deps não reinstalam.

6. Imagem vs contêiner. build = molde imutável (classe); run = instância viva (objeto).

7. Compose. Vários serviços em 1 YAML; rede criada automaticamente; API acessa postgres pelo nome.

8. Produção. CD → build → registry (Hub/ECR) → tags → rollback instantâneo. Imagem imutável = paridade dev/prod.

+ Limitações. Segurança (kernel compartilhado), persistência (volumes), complexidade para app simples, rede difícil de debugar.

Exemplos de código — Docker na prática

3 exemplos (Python 3.14, Java 26, Go 1.26), validados por execução.

1. Dockerfile + docker-compose.yml

# Dockerfile
FROM node:20-alpine
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
EXPOSE 3000
CMD ["npm", "start"]
# docker-compose.yml (3 servicos)
services:
  api:
    build: .
    ports:
      - "3000:3000"
    depends_on:
      - postgres
    environment:
      - DATABASE_URL=postgres://user:pass@postgres:5432/app
  postgres:
    image: postgres:16
    environment:
      - POSTGRES_PASSWORD=pass
    volumes:
      - pgdata:/var/lib/postgresql/data
  redis:
    image: redis:7
    ports:
      - "6379:6379"
volumes:
  pgdata:
# compose: 1 comando sobe tudo
# $ docker compose up
# - cria a rede automaticamente
# - api + postgres + redis
# - api acessa postgres pelo nome
#   (sem configurar IP)
# - volume pgdata persiste os dados

2. Camadas cacheadas — build de verdade

# Python 3.14 — cache de camadas
import hashlib, json, os

CACHE = '/tmp/docker_layers.json'

def restore():
    if os.path.exists(CACHE):
        return json.load(
          open(CACHE))
    return {}

def build(deps, code):
    cache = restore()
    h = hashlib.sha256(
      deps.encode()).hexdigest()
    if cache.get(
      'deps_hash') == h:
        print('deps OK (cache hit)')
    else:
        print('deps mudaram — reinstall')
        cache['deps_hash'] = h
    print(f'copiando {code}')
    json.dump(cache,
      open(CACHE, 'w'))

build('express@4', 'app v1')
build('express@4', 'app v2')
# 1o: instala deps; 2o: cache hit
// Java 26 — camadas/cache
import java.nio.file.*;
import java.util.*;

public class LayersDemo {
  public static void main(
    String[] a)
    throws Exception {
    var p =
      Path.of("/tmp/layers.txt");
    var cache =
      Files.exists(p) ?
      Files.readString(p) :
      "";
    String deps =
      "react@19";
    if (cache.equals(deps)) {
      System.out.println(
        "deps OK (cache hit)");
    } else {
      System.out.println(
        "deps mudaram — reinstall");
      Files.writeString(p, deps);
    }
    System.out.println(
      "copiando codigo...");
  }
}
// Go 1.26 — camadas/cache
package main

import (
    "fmt"
    "os"
)

func main() {
    const deps =
      "express@4"
    b, _ := os.ReadFile(
      "/tmp/go_layers.txt")
    if string(b) == deps {
        fmt.Println(
          "deps OK (cache hit)")
    } else {
        fmt.Println(
        "deps mudaram — reinstall")
        os.WriteFile(
          "/tmp/go_layers.txt",
          []byte(deps), 0o644)
    }
    fmt.Println(
      "copiando codigo...")
}

3. Namespaces simulados (o "PID 1" de cada contêiner)

# Python 3.14
class Namespace:
    def __init__(self, nome):
        self.nome = nome
        self.pid = 1
        self.processos = []

ns_web = Namespace('web')
ns_web.processos.append(
  'node server.js')

ns_db = Namespace('db')
ns_db.processos.append(
  'postgres -D /var/lib')

print(f"{ns_web.nome}: acha PID "
  f"{ns_web.pid} (host: 2451)")
print(f"{ns_db.nome}: acha PID "
  f"{ns_db.pid} (host: 3120)")
# ilusao de isolamento
// Java 26
import java.util.*;

public class NamespaceDemo {
  record NS(String nome,
    int pid,
    List<String> procs) {}
  public static void main(
    String[] a) {
    var web = new NS(
      "web", 1,
      List.of(
        "node server.js"));
    var db = new NS(
      "db", 1,
      List.of(
        "postgres -D /var/lib"));
    System.out.println(
      web.nome() + ": acha PID "
      + web.pid() +
      " (host: 2451)");
    System.out.println(
      db.nome() + ": acha PID "
      + db.pid() +
      " (host: 3120)");
  }
}
// Go 1.26
package main

import "fmt"

type NS struct {
    nome string
    pid  int
}

func main() {
    web := NS{"web", 1}
    db := NS{"db", 1}
    fmt.Printf("%s: acha PID %d (host: 2451)\n",
      web.nome, web.pid)
    fmt.Printf("%s: acha PID %d (host: 3120)\n",
      db.nome, db.pid)
}
