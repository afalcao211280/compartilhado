Resumo — "8 conceitos que fazem um Banco de Dados funcionar"

Fonte: youtube.com/watch?v=wNM8wETFss4 · pt-BR · ~11:40 · 30/08/2026

Tese central: você confirma um Pix e em milissegundos o banco muda bytes na memória, grava um log no disco e promete que o dado não vai sumir. Como uma mensagem do back end vira dado persistido, por dentro.

Pontos-chave

1. Processo + conexão. SELECT lê; INSERT/UPDATE/DELETE muda estado. Por baixo: achar dados, controlar quem vê/muda, sobreviver a falhas.

2. Páginas. Banco não busca linha solta — usa blocos (páginas). Linhas espalhadas = lento; próximas = barato. Modelagem + índices definem o caminho.

3. Buffer pool. Cache interno de páginas. Hit = na memória; miss = disco. Primeira query mais lenta, depois rápida. Dirty page: memória mudou, disco ainda não.

4. WAL. Grava a mudança no log antes da página final. Commit OK depende do log, não do arquivo de dados.

5. Durabilidade. Após queda, replay do log: refaz confirmadas, descarta não confirmadas.

6. Transações. BEGIN/COMMIT/ROLLBACK — todas as mudanças juntas ou nada.

7. Concorrência. Locks travam dados; MVCC mantém versões antigas (leitura não bloqueia escrita). Read Committed vs Repeatable Read = qual versão você vê.

8. Índices + manutenção. B-tree acha sem varrer páginas; mas custa escrita. Vacuum/checkpoint/recovery são o trabalho de bastidor.

Exemplos de código — os 3 pilares

3 exemplos (Python 3.14, Java 26, Go 1.26), sem dependências, validados por execução.

1. WAL: log antes do commit

# Python 3.14 — write-ahead log
import json, os

LOG = '/tmp/wal.log'

def write_wal(tx_id, entries):
    """Grava ANTES de mudar o
    'disco' (arquivo de dados)."""
    with open(LOG, 'a') as f:
        f.write(json.dumps({'tx': tx_id,
            'entries': entries}) + '\n')

def apply(entries):
    """Aplica no arquivo de dados
    (pode ser depois)."""
    data = {}
    if os.path.exists(
        '/tmp/data.json'):
        data = json.load(open(
            '/tmp/data.json'))
    for k, v in entries.items():
        data[k] = data.get(k, 0) + v
    json.dump(data, open(
        '/tmp/data.json', 'w'))

def commit(tx_id, entries):
    write_wal(tx_id, entries) # 1
    apply(entries)             # 2
    return 'commit OK'

# simula queda entre log e apply
write_wal('tx-1',
  {'a': -100, 'b': 100})
print('caiu antes do apply —'
  ' log tem o registro')
print(open(LOG).read().strip())
// Java 26 — WAL
import java.nio.file.*;
import java.util.*;

public class WalDemo {
  static void wal(String tx,
      String entries)
      throws Exception {
    // grava ANTES
    Files.writeString(
        Path.of("/tmp/wal.log"),
        tx + " " + entries +
        "\n",
        StandardOpenOption.CREATE,
        StandardOpenOption.APPEND);
  }
  public static void main(
      String[] a)
      throws Exception {
    wal("tx-1",
      "{\"a\":-100,\"b\":100}");
    wal("tx-2",
      "{\"a\":50,\"b\":-50}");
    System.out.println(
      "log (replay na subida):");
    System.out.println(
      Files.readString(
        Path.of(
          "/tmp/wal.log")));
  }
}
// Go 1.26 — WAL
package main

import (
    "fmt"
    "os"
)

func wal(tx, entry string) {
    f, _ := os.OpenFile(
        "/tmp/wal.log",
        os.O_APPEND|os.O_CREATE|
        os.O_WRONLY, 0o644)
    defer f.Close()
    fmt.Fprintf(f, "%s %s\n",
        tx, entry)
}

func main() {
    wal("tx-1",
      `{"a":-100,"b":100}`)
    wal("tx-2",
      `{"a":50,"b":-50}`)
    fmt.Println("log (replay:")
    b, _ := os.ReadFile(
        "/tmp/wal.log")
    fmt.Print(string(b))
}

2. Transação: tudo ou nada

# Python 3.14
class Transacao:
    def __init__(self, saldos):
        self.saldos = saldos
        self._backup = None
    def begin(self):
        self._backup = dict(self.saldos)
    def debitar(self, c, v):
        if self.saldos[c] < v:
            raise ValueError(
                'saldo insuficiente')
        self.saldos[c] -= v
    def creditar(self, c, v):
        self.saldos[c] += v
    def commit(self):
        self._backup = None
    def rollback(self):
        self.saldos = dict(self._backup)

t = Transacao(
    {'a': 100, 'b': 0})
t.begin()
t.debitar('a', 100)
t.creditar('b', 100)
t.commit()
print(t.saldos)
# {'a': 0, 'b': 100}

t.begin()
try:
    t.debitar('a', 999)
except ValueError:
    t.rollback()
print(t.saldos)
# volta ao estado anterior
// Java 26
import java.util.*;

public class TxDemo {
  static class Tx {
    HashMap<String, Integer> saldos;
    HashMap<String, Integer> backup = null;
    Tx(HashMap<String, Integer> s) { saldos = s; }
    void begin() {
      backup = new HashMap<>(saldos); }
    void debitar(String c, int v) {
      if (saldos.get(c) < v)
        throw new IllegalArgumentException();
      saldos.put(c, saldos.get(c) - v); }
    void creditar(
      String c, int v) {
      saldos.put(c,
        saldos.get(c) + v); }
    void commit() {
      backup = null; }
    void rollback() {
      saldos.clear();
      saldos.putAll(backup); }
  }
  public static void main(
    String[] a) {
    var t = new Tx(
      new HashMap<>(
        Map.of("a", 100,
          "b", 0)));
    t.begin(); t.debitar("a", 100);
    t.creditar("b", 100);
    t.commit();
    System.out.println(
      t.saldos);
    t.begin();
    try { t.debitar(
      "a", 999); }
    catch (IllegalArgumentException
      e) { t.rollback(); }
    System.out.println(
      t.saldos);
  }
}
// Go 1.26
package main

import (
    "fmt"
    "log"
)

type Tx struct {
    saldos map[string]int
    backup map[string]int
}

func (t *Tx) begin() {
    t.backup = clone(t.saldos) }
func (t *Tx) debitar(
    c string, v int) {
    if t.saldos[c] < v {
        log.Fatal(
        "saldo insuficiente")
    }
    t.saldos[c] -= v }
func (t *Tx) creditar(
    c string, v int) {
    t.saldos[c] += v }
func (t *Tx) commit() {
    t.backup = nil }
func (t *Tx) rollback() {
    t.saldos = clone(t.backup) }

func clone(m map[string]int) map[string]int {
    out := make(
      map[string]int,
      len(m))
    for k, v := range m {
        out[k] = v }
    return out
}

func main() {
    t := &Tx{saldos:
      map[string]int{
        "a": 100, "b": 0}}
    t.begin()
    t.debitar("a", 100)
    t.creditar("b", 100)
    t.commit()
    fmt.Println(t.saldos)
    t.begin()
    t.rollback()
    fmt.Println(t.saldos)
}

3. MVCC: leitura não bloqueia escrita

# Python 3.14 — MVCC simulado
class MVCC:
    def __init__(self):
        self.versions = {}
    def write(self, key,
      value, tx_id):
        self.versions.setdefault(
          key, []).append(
            (value, tx_id))
    def read(self, key, tx_id):
        for value, wtx in reversed(
            self.versions[key]):
            if wtx <= tx_id:
                return value
        return None

db = MVCC()
db.write('saldo', 100, 1)
db.write('saldo', 90, 2)
print(db.read(
  'saldo', 1)) # 100
print(db.read(
  'saldo', 2)) # 90
// Java 26 — MVCC simulado
import java.util.*;

public class MvccDemo {
  record Ver(
    int value, int tx) {}
  static Map<String,
    List<Ver>> versions =
    new HashMap<>();

  static void write(
    String k, int v,
    int tx) {
    versions.computeIfAbsent(k,
      x -> new ArrayList<>())
      .add(new Ver(v, tx));
  }
  static Integer read(
    String k, int tx) {
    var list = versions.get(k);
    if (list == null)
      return null;
    for (int i =
      list.size() - 1; i >= 0; i--)
      if (list.get(i).tx() <= tx)
        return
          list.get(i).value();
    return null;
  }
  public static void main(
    String[] a) {
    write("saldo", 100, 1);
    write("saldo", 90, 2);
    System.out.println(
      read("saldo", 1));
    System.out.println(
      read("saldo", 2));
  }
}
// Go 1.26 — MVCC simulado
package main

import "fmt"

type Ver struct {
    value int
    tx    int
}

var versions =
  map[string][]Ver{}

func write(
    k string, v, tx int) {
    versions[k] = append(
      versions[k], Ver{v, tx})
}

func read(k string,
    tx int) (int, bool) {
    list := versions[k]
    for i := len(list) - 1;
        i >= 0; i-- {
        if list[i].tx <= tx {
            return list[i].value, true
        }
    }
    return 0, false
}

func main() {
    write("saldo", 100, 1)
    write("saldo", 90, 2)
    fmt.Println(read(
      "saldo", 1)) // 100
    fmt.Println(read(
      "saldo", 2)) // 90
}
